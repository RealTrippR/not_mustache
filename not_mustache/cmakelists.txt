cmake_minimum_required(VERSION 3.23)
project(not_mustache_lib)

add_library(not_mustache_lib STATIC src/not_mustache.c)
target_include_directories(not_mustache_lib PUBLIC  ${CMAKE_CURRENT_SOURCE_DIR}/deps  ${CMAKE_CURRENT_SOURCE_DIR}/inc)


if (WIN32)
    target_sources(not_mustache_lib INTERFACE src/not_mustache.c deps/streql/streql_x64_win.obj deps/streql/strneql_x64_win.obj)
elseif (UNIX)
    target_sources(not_mustache_lib INTERFACE src/not_mustache.c deps/streql/streql_x64_unix.o deps/streql/strneql_x64_unix.o)
else()
    target_sources(not_mustache_lib INTERFACE src/not_mustache.c)
endif()

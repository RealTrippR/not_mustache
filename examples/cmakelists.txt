cmake_minimum_required(VERSION 3.23)
project(not_mustache_examples)

get_target_property(msrc not_mustache_lib SOURCE_DIR)
file(GLOB EXAMPLE_SOURCES ${CMAKE_CURRENT_SOURCE_DIR}/*.c)

foreach(example ${EXAMPLE_SOURCES})

    get_filename_component(example_name ${example} NAME_WE)

    add_executable(${example_name} ${example})
    target_include_directories(${example_name} PUBLIC ${msrc}/inc  ${CMAKE_CURRENT_SOURCE_DIR})
    target_link_libraries(${example_name}
        PUBLIC
            not_mustache_lib
    )

endforeach()
